if(Git_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" rev-parse --show-toplevel
    OUTPUT_VARIABLE _git_top OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(_git_top STREQUAL PROJECT_SOURCE_DIR)
    foreach(_git_name HEAD packed-refs)
      execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" rev-parse --git-path "${_git_name}"
        OUTPUT_VARIABLE _git_file OUTPUT_STRIP_TRAILING_WHITESPACE)
      get_filename_component(_git_file "${_git_file}" ABSOLUTE BASE_DIR "${PROJECT_SOURCE_DIR}")
      if(EXISTS "${_git_file}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_git_file}")
      endif()
    endforeach()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" symbolic-ref -q HEAD
      OUTPUT_VARIABLE _git_branch OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(NOT _git_branch STREQUAL "")
      execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" rev-parse --git-path "${_git_branch}"
        OUTPUT_VARIABLE _git_ref OUTPUT_STRIP_TRAILING_WHITESPACE)
      get_filename_component(_git_ref "${_git_ref}" ABSOLUTE BASE_DIR "${PROJECT_SOURCE_DIR}")
      if(EXISTS "${_git_ref}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_git_ref}")
      endif()
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" rev-parse --git-common-dir
      OUTPUT_VARIABLE _git_common OUTPUT_STRIP_TRAILING_WHITESPACE)
    get_filename_component(_git_common "${_git_common}" ABSOLUTE BASE_DIR "${PROJECT_SOURCE_DIR}")
    file(GLOB_RECURSE _tag_refs CONFIGURE_DEPENDS "${_git_common}/refs/tags/*")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_tag_refs})
  endif()
endif()

function(jaro_executable target)
  add_executable(${target} ${ARGN})
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
  if(JARO_STATIC)
    target_link_options(${target} PRIVATE -static)
  endif()
endfunction()
