# Build the complete, pinned release sources without running upstream Autotools.
set(_source "${PROJECT_SOURCE_DIR}/vendor/abook/upstream")
set(JARO_ABOOK_DEPS_PREFIX "" CACHE PATH "Target ncurses/readline prefix for abook (required for static cross builds)")
option(JARO_ABOOK_NLS "Build abook message translations (requires gettext tools and libintl)" OFF)
if(JARO_STATIC AND NOT JARO_ABOOK_DEPS_PREFIX)
  message(FATAL_ERROR "Static abook needs target ncurses/readline: run build/abook-deps.py and set JARO_ABOOK_DEPS_PREFIX")
endif()
if(JARO_ABOOK_DEPS_PREFIX)
  # Never fall back to host libraries in a cross build.
  set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
  find_path(ABOOK_CURSES_INCLUDE_DIR ncurses.h PATHS "${JARO_ABOOK_DEPS_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(ABOOK_CURSES_LIBRARY ncursesw PATHS "${JARO_ABOOK_DEPS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  find_path(ABOOK_READLINE_INCLUDE_DIR readline/readline.h PATHS "${JARO_ABOOK_DEPS_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(ABOOK_READLINE_LIBRARY readline PATHS "${JARO_ABOOK_DEPS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  set(_curses_libraries "${ABOOK_CURSES_LIBRARY}")
else()
  set(CURSES_NEED_NCURSES TRUE)
  set(CURSES_NEED_WIDE TRUE)
  find_package(Curses REQUIRED)
  set(ABOOK_CURSES_INCLUDE_DIR "${CURSES_INCLUDE_DIRS}")
  set(_curses_libraries ${CURSES_LIBRARIES})
  find_path(ABOOK_READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
  find_library(ABOOK_READLINE_LIBRARY readline REQUIRED)
endif()

include(CheckIncludeFile)
include(CheckIncludeFiles)
include(CheckSymbolExists)
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
set(CMAKE_REQUIRED_INCLUDES ${ABOOK_CURSES_INCLUDE_DIR} "${ABOOK_READLINE_INCLUDE_DIR}")
set(CMAKE_REQUIRED_LIBRARIES "${ABOOK_READLINE_LIBRARY}" ${_curses_libraries})
foreach(_header unistd.h locale.h sys/ioctl.h stdarg.h wchar.h wctype.h ncurses.h)
  string(MAKE_C_IDENTIFIER "HAVE_${_header}" _variable)
  string(TOUPPER "${_variable}" _variable)
  check_include_file("${_header}" ${_variable})
endforeach()
check_include_files("stdio.h;readline/readline.h" HAVE_READLINE_READLINE_H)
check_include_files("stdio.h;readline/history.h" HAVE_READLINE_HISTORY_H)
foreach(_function setlocale snprintf vsnprintf strcasestr strcoll mbtowc wcwidth mbrtowc mbsinit iswprint iswcntrl)
  string(TOUPPER "HAVE_${_function}" _variable)
  check_symbol_exists(${_function} "stdlib.h;stdio.h;string.h;locale.h;wchar.h;wctype.h" ${_variable})
endforeach()
check_symbol_exists(resizeterm "ncurses.h" HAVE_RESIZETERM)
check_symbol_exists(add_history "stdio.h;readline/history.h" HAVE_READLINE_HISTORY)
if(NOT HAVE_READLINE_HISTORY)
  message(FATAL_ERROR "abook needs GNU readline with history support")
endif()
set(HAVE_DECL_WCWIDTH "${HAVE_WCWIDTH}")
if(HAVE_WCHAR_H AND HAVE_MBRTOWC AND HAVE_MBSINIT AND HAVE_MBTOWC AND HAVE_WCWIDTH)
  set(HANDLE_MULTIBYTE 1)
endif()
if(JARO_ABOOK_NLS)
  find_package(Intl REQUIRED)
  find_package(Gettext REQUIRED)
  set(ENABLE_NLS 1)
endif()
configure_file(config.h.in generated/config.h)
set(_sources abook.c abook_rl.c database.c edit.c filter.c getname.c getopt.c getopt1.c gettext.c
  ldif.c list.c mbswidth.c misc.c options.c remap.c ui.c views.c xmalloc.c)
list(TRANSFORM _sources PREPEND "${_source}/")
jaro_executable(abook ${_sources})
set_target_properties(abook PROPERTIES C_EXTENSIONS ON)
target_compile_definitions(abook PRIVATE HAVE_CONFIG_H=1 _GNU_SOURCE)
target_compile_options(abook PRIVATE "$<$<C_COMPILER_ID:GNU,Clang,AppleClang>:-fgnu89-inline>")
target_include_directories(abook PRIVATE "${_source}" "${CMAKE_CURRENT_BINARY_DIR}/generated"
  ${ABOOK_CURSES_INCLUDE_DIR} "${ABOOK_READLINE_INCLUDE_DIR}")
target_link_libraries(abook PRIVATE "${ABOOK_READLINE_LIBRARY}" ${_curses_libraries})
if(JARO_ABOOK_NLS)
  target_link_libraries(abook PRIVATE Intl::Intl)
  foreach(_language de fr it ja sv)
    set(_catalog "${CMAKE_CURRENT_BINARY_DIR}/locale/${_language}/LC_MESSAGES/abook.mo")
    add_custom_command(OUTPUT "${_catalog}"
      COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/locale/${_language}/LC_MESSAGES"
      COMMAND "${GETTEXT_MSGFMT_EXECUTABLE}" -o "${_catalog}" "${_source}/po/${_language}.po"
      DEPENDS "${_source}/po/${_language}.po" VERBATIM)
    list(APPEND _catalogs "${_catalog}")
    install(FILES "${_catalog}" DESTINATION "${CMAKE_INSTALL_LOCALEDIR}/${_language}/LC_MESSAGES")
  endforeach()
  add_custom_target(abook-translations ALL DEPENDS ${_catalogs})
  add_dependencies(abook abook-translations)
endif()
install(TARGETS abook RUNTIME DESTINATION "${JARO_RUNTIME_DIR}/bin")
install(FILES "${_source}/abook.1" DESTINATION "${CMAKE_INSTALL_MANDIR}/man1")
install(FILES "${_source}/abookrc.5" DESTINATION "${CMAKE_INSTALL_MANDIR}/man5")
install(FILES "${_source}/sample.abookrc" DESTINATION "${JARO_RUNTIME_DIR}/abook")
install(FILES "${_source}/COPYING" "${_source}/AUTHORS" DESTINATION "${JARO_RUNTIME_DIR}/licenses/abook")
if(JARO_STATIC)
  install(DIRECTORY "${JARO_ABOOK_DEPS_PREFIX}/share/licenses/" DESTINATION "${JARO_RUNTIME_DIR}/licenses/static-deps")
endif()
